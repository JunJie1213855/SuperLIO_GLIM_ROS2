# ROS 2 interfaces are built as part of the glim package.
option(BUILD_WITH_CV_BRIDGE "Enable camera image support in ROS 2" ON)
set(glim_ros_dependencies
  rclcpp rclcpp_components rosbag2_cpp rosbag2_compression rosbag2_storage
  image_transport nav_msgs sensor_msgs geometry_msgs tf2_ros)
foreach(dependency IN LISTS glim_ros_dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
find_package(Boost REQUIRED COMPONENTS program_options)
if(NOT BUILD_WITH_OPENCV)
  set(BUILD_WITH_CV_BRIDGE OFF)
endif()
if(BUILD_WITH_CV_BRIDGE)
  find_package(cv_bridge REQUIRED)
  list(APPEND glim_ros_dependencies cv_bridge)
  add_compile_definitions(BUILD_WITH_CV_BRIDGE)
endif()
if(BUILD_WITH_CUDA)
  add_compile_definitions(BUILD_GTSAM_POINTS_GPU)
endif()
set(OLD_DISTRO humble galactic foxy)
if("$ENV{ROS_DISTRO}" IN_LIST OLD_DISTRO)
  add_compile_definitions(CV_BRIDGE_INCLUDE_H)
else()
  add_compile_definitions(CV_BRIDGE_INCLUDE_HPP)
endif()
add_library(glim_ros SHARED src/glim_ros/glim_ros.cpp src/glim_ros/ros_qos.cpp)
target_include_directories(glim_ros PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include/glim>)
ament_target_dependencies(glim_ros ${glim_ros_dependencies})
target_link_libraries(glim_ros glim)
add_library(rviz_viewer SHARED src/glim_ros/rviz_viewer.cpp)
target_include_directories(rviz_viewer PRIVATE include)
ament_target_dependencies(rviz_viewer ${glim_ros_dependencies})
target_link_libraries(rviz_viewer glim)
foreach(node glim_rosnode glim_rosbag validator_node)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} ${glim_ros_dependencies})
  target_link_libraries(${node} glim_ros)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_WITH_VIEWER)
  foreach(viewer offline_viewer map_editor)
    # map_editor is also the name of a core shared-library target.
    add_executable(${viewer}_ros src/${viewer}.cpp)
    set_target_properties(${viewer}_ros PROPERTIES OUTPUT_NAME ${viewer})
    ament_target_dependencies(${viewer}_ros ${glim_ros_dependencies})
    target_include_directories(${viewer}_ros PRIVATE include)
    if(viewer STREQUAL "offline_viewer")
      target_link_libraries(${viewer}_ros interactive_viewer Boost::program_options)
    else()
      target_link_libraries(${viewer}_ros map_editor Boost::program_options)
    endif()
    install(TARGETS ${viewer}_ros DESTINATION lib/${PROJECT_NAME})
  endforeach()
endif()
install(TARGETS glim_ros rviz_viewer LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include/glim)
install(DIRECTORY rviz DESTINATION share/${PROJECT_NAME})
